cmake_minimum_required(VERSION 3.24)
project(TurboHyMT LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED HYMT_LLAMA_SOURCE)
  message(FATAL_ERROR "Set HYMT_LLAMA_SOURCE to pinned STQ checkout")
endif()
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${HYMT_LLAMA_SOURCE}" OUTPUT_VARIABLE LLAMA_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT LLAMA_COMMIT STREQUAL "1e411d8f5a1e23525fa3265dfb4bd76265465397")
  message(FATAL_ERROR "Unexpected STQ source revision")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod" CACHE STRING "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
add_subdirectory("${HYMT_LLAMA_SOURCE}" llama-build)
add_library(TurboHyMT STATIC HyMTBridge.cpp)
target_include_directories(TurboHyMT PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(TurboHyMT PRIVATE llama)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(hymt-tests HyMTBridgeTests.cpp)
  target_link_libraries(hymt-tests PRIVATE TurboHyMT)
endif()
